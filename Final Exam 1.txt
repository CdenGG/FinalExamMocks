19 (13 points) Create a JavaFX application start method that displays a simple calculator interface using a GridPane.

    Row 0: A TextField (read-only for display) spanning 4 columns.
    Row 1: Buttons "7", "8", "9", "/"
    Row 2: Buttons "4", "5", "6", "*"
    Row 3: Buttons "1", "2", "3", "-"
    Row 4: Buttons "0", ".", "=", "+"
    Set reasonable spacing/padding. Create the scene and show the stage. No event handling logic required for this problem. Focus only on the layout setup.

Java

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.layout.GridPane;
import javafx.stage.Stage;
// Potentially: import javafx.scene.layout.Priority;

public class CalculatorLayout extends Application {
    @Override
    public void start(Stage primaryStage) {
        GridPane grid = new GridPane();
        grid.setPadding(new Insets(10));
        grid.setHgap(5);
        grid.setVgap(5);

        TextField display = new TextField();
        display.setEditable(false);
        display.setAlignment(javafx.geometry.Pos.CENTER_RIGHT);
        grid.add(display, 0, 0, 4, 1); // Span 4 columns

        String[] buttons = {
            "7", "8", "9", "/",
            "4", "5", "6", "*",
            "1", "2", "3", "-",
            "0", ".", "=", "+"
        };

        int row = 1, col = 0;
        for (String text : buttons) {
            Button btn = new Button(text);
            // Optional: btn.setMaxWidth(Double.MAX_VALUE); // Make buttons fill width
            // Optional: GridPane.setHgrow(btn, Priority.ALWAYS); // Make buttons grow
            grid.add(btn, col, row);
            col++;
            if (col > 3) {
                col = 0;
                row++;
            }
        }

        Scene scene = new Scene(grid);
        primaryStage.setTitle("Calculator Layout");
        primaryStage.setScene(scene);
        primaryStage.show();
    }
    // main method...
}
