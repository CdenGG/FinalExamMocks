import javafx.geometry.Insets;

import javafx.scene.layout.Pane;
import javafx.stage.Stage;
import java.util.Locale; // For formatting

public class MouseTrackerApp extends Application {
    @Override
    public void start(Stage primaryStage) {
        Pane root = new Pane();
        root.setPadding(new Insets(10));

        Label coordinatesLabel = new Label("X: ---, Y: ---");
        coordinatesLabel.relocate(10, 10); // Position label
        root.getChildren().add(coordinatesLabel);

        root.setOnMouseMoved(event -> {
            double x = event.getX();
            double y = event.getY();
            // Format the string (e.g., using String.format)

            // Update label text
        });

        Scene scene = new Scene(root, 400, 300);
        primaryStage.setTitle("Mouse Tracker");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

}
